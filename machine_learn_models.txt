def wrap_with_log_target(estimator, param_dist: dict):
    
    wrapped = TransformedTargetRegressor(
        regressor=estimator,
        func=np.log1p,
        inverse_func=np.expm1,
        check_inverse=False
    )
    new_param_dist = {f"regressor__{k}": v for k, v in (param_dist or {}).items()}
    return wrapped, new_param_dist

def unwrap_pipeline(maybe_wrapped):
    
    if isinstance(maybe_wrapped, TransformedTargetRegressor):
        return getattr(maybe_wrapped, "regressor_", maybe_wrapped.regressor)
    return maybe_wrapped

def preprocess_time(df: pd.DataFrame) -> pd.DataFrame:
    df = df.copy()
    df.columns = df.columns.str.strip()

    if "Ağırlık (gr)" in df.columns:
        wcol = "Ağırlık (gr)"
        df[wcol] = (
            df[wcol]
            .astype(str)
            .str.replace(".", "", regex=False)
            .str.replace(",", ".", regex=False)
        )
        df[wcol] = pd.to_numeric(df[wcol], errors="coerce")

    for c in list(df.columns):
        if str(c).startswith("Unnamed"):
            df = df.drop(columns=c)

    COL_DATE  = "Date"
    COL_START = "Start"
    COL_END   = "Finish"

    df[COL_DATE] = pd.to_datetime(df[COL_DATE], errors="coerce")
    def fix_time(s):
        s = str(s).strip()
        if s == "" or s.lower() == "nan":
            return np.nan
        if s.count(":") == 1:
            return s + ":00"
        return s

    df[COL_START] = df[COL_START].fillna("00:00").astype(str).map(fix_time)
    df[COL_END]   = df[COL_END].fillna("00:00").astype(str).map(fix_time)

    df["start_td"] = pd.to_timedelta(df[COL_START], errors="coerce")
    df["end_td"]   = pd.to_timedelta(df[COL_END],   errors="coerce")

    df["start_dt"] = df[COL_DATE] + df["start_td"]
    df["end_dt"]   = df[COL_DATE] + df["end_td"]

    df = df.dropna(subset=["start_dt", "end_dt"])
    print("-> start_dt / end_dt NaN atıldıktan sonra:", len(df))

    mask_next_day = df["end_dt"] < df["start_dt"]
    df.loc[mask_next_day, "end_dt"] = df.loc[mask_next_day, "end_dt"] + pd.Timedelta(days=1)

    df["window_min"] = (df["end_dt"] - df["start_dt"]).dt.total_seconds() / 60.0

    return df

def train_and_eval(model_name, estimator, param_dist, use_ann_preprocessor=False):
    chosen_prep = preprocessor_ann if use_ann_preprocessor else preprocessor_tree

    pipe = Pipeline(
        steps=[
            ("prep", chosen_prep),
            ("model", estimator),
        ]
    )

    search_estimator = pipe
    search_param_dist = param_dist

    if USE_LOG_TARGET_SHORT:
        search_estimator, search_param_dist = wrap_with_log_target(pipe, param_dist)

    search = RandomizedSearchCV(
        estimator=search_estimator,
        param_distributions=search_param_dist,
        n_iter=20,
        cv=3,
        scoring="neg_root_mean_squared_error",
        random_state=42,
        n_jobs=-1,
        verbose=1,
    )

  
    search.fit(X_train, y_train)
    best_model = search.best_estimator_
    print(f"{model_name} en iyi hiperparametreler:", search.best_params_)

    y_pred = best_model.predict(X_test) 

    rmse = mean_squared_error(y_test, y_pred) ** 0.5
    mae = mean_absolute_error(y_test, y_pred)

    mask_nonzero = y_test != 0
    if mask_nonzero.sum() > 0:
        mape = (
            np.abs((y_test[mask_nonzero] - y_pred[mask_nonzero]) / y_test[mask_nonzero])
        ).mean() * 100
    else:
        mape = np.nan

    r2 = r2_score(y_test, y_pred)

    print(
        f"{model_name} | RMSE={rmse:.4f} dk  "
        f"MAE={mae:.4f} dk  "
        f"MAPE={mape:.2f}%  R2={r2:.4f}"
    )

    return {
        "name": model_name,
        "best_model": best_model,
        "rmse": rmse,
        "mae": mae,
        "mape": mape,
        "r2": r2,
        "y_pred": pd.Series(y_pred, index=X_test.index),
    }

def build_feature_row_for_part(
    part_code,
    machine=None,
    mold=None,
    qty=None,
    goz_sayisi=None,
    vardiya=None,
    hammadde=None,
):
    subset = df[df["Component"] == part_code]
    values = {}

    for col in numeric_features + categorical_features:
        if col in subset and not subset[col].isna().all():
            if col in numeric_features:
                values[col] = subset[col].median()
            else:
                mode_series = subset[col].mode()
                values[col] = (
                    mode_series.iloc[0]
                    if not mode_series.empty
                    else global_defaults[col]
                )
        else:
            values[col] = global_defaults[col]

    if machine is not None and "Machine" in categorical_features:
        values["Machine"] = machine
    if mold is not None and "Device" in categorical_features:
        values["Device"] = mold
    if qty is not None and "Quantity" in numeric_features:
        values["Quantity"] = qty
    if por is not None and "Portion" in categorical_features:
        values["Portion"] = por
    if per is not None and "Period" in categorical_features:
        values["Period"] = per
    if raw is not None and "Raw" in categorical_features:
        values["Hammadde Kodu"] = raw

    X_new = pd.DataFrame([values])

    for col in categorical_features:
        if col in X_new.columns:
            X_new[col] = X_new[col].astype("string")

    return X_new, values
